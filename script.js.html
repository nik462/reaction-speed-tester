let box = document.getElementById("box"); let result = document.getElementById("result"); let startTime; let timeout; box.addEventListener("click", () => { if (box.style.backgroundColor === "green") { let reaction = Date.now() - startTime; result.textContent = "Your reaction time: " + reaction + " ms"; reset(); } }); function reset() { box.style.backgroundColor = "red"; box.textContent = "Wait for green..."; result.textContent = ""; start(); } function start() { let delay = Math.random() * 3000 + 1000; timeout = setTimeout(() => { box.style.backgroundColor = "green"; box.textContent = "CLICK!"; startTime = Date.now(); }, delay); } start();
